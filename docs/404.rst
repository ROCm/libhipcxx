          document.getElementById('linkGen').innerHTML = ""
          result.forEach(element => {
            document.getElementById('linkGen').innerHTML += listItem.replace(/\{0\}/g, element.item).replace(/\{1\}/g, element.score.toFixed(2))
          });
        })
    </script>
